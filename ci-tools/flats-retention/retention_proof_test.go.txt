package notebook

import (
	"os"
	"testing"
)

var retentionProbeAttempt int
var retentionProbeBuffers [][]byte

// This disposable adapter supplies retained bytes independently of the sampler.
func retentionProbeRead(t *testing.T, in *Instance, query []byte) ([]byte, error) {
	body, err := in.Read(t.Context(), query, false)
	if err != nil {
		return body, err
	}
	retentionProbeAttempt++
	mode := os.Getenv("RETENTION_PROBE")
	size := 0
	switch mode {
	case "first":
		if retentionProbeAttempt == 1 {
			size = 2 << 20
		}
	case "middle":
		if retentionProbeAttempt == 2 {
			size = 2 << 20
		}
	case "last":
		if retentionProbeAttempt == 3 {
			size = 2 << 20
		}
	case "per_read":
		size = 400 << 10
	case "response":
		owned := make([]byte, 2<<20)
		copy(owned, body)
		body = owned[:len(body)]
	default:
		t.Fatalf("unexpected retention probe mode %q", mode)
	}
	if size != 0 {
		buffer := make([]byte, size)
		buffer[0], buffer[len(buffer)-1] = 1, 1
		retentionProbeBuffers = append(retentionProbeBuffers, buffer)
	}
	t.Logf("RETENTION_PROBE mode=%s attempt=%d retained=%d response_len=%d response_cap=%d", mode, retentionProbeAttempt, size, len(body), cap(body))
	return body, nil
}
